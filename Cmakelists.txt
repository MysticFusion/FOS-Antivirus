# 1. Specify the minimum version of CMake required
cmake_minimum_required(VERSION 3.10)

# 2. Name your project and specify the language
project(FOS_Antivirus C)

include(CTest)

# --- NEW: Find PkgConfig, GTK, and SQLite ---
find_package(PkgConfig REQUIRED)
# Note: If your code is specifically written for GTK4
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4)
find_package(SQLite3 REQUIRED)
# -----------------------------------

# 3. Organize source files into variables for readability
set(MAIN_SOURCE src/main.c)

set(APP_SOURCES
    src/app/app.c
    src/app/app_paths.c
    src/app/auto_update.c
)

set(UTIL_SOURCES
    src/utils/path_utils.c
)

set(ENGINE_SOURCES
    src/engine/ed25519_verify.c
    src/engine/feature_extract.c
    src/engine/fs_enumerator.c
    src/engine/hash_util.c
    src/engine/heuristic_engine.c
    src/engine/ml_engine.c
    src/engine/response_engine.c
    src/engine/rt_monitor.c
    src/engine/scan_core.c
    src/engine/scan_executor.c
    src/engine/scan_persistence.c
    src/engine/scan_processes.c
    src/engine/scan_progress.c
    src/engine/scan_report_bridge.c
    src/engine/sha2.c
    src/engine/signature_scan.c
    src/engine/signature_scan_sqlite.c
    src/engine/trust.c
)

set(UI_SOURCES
    src/ui/ui_history.c
    src/ui/ui_scan_paths.c
    src/ui/ui_scan.c
    src/ui/ui_sidebar.c
    src/ui/ui_update.c
    src/ui/ui_views.c
)

# 4. Tell CMake to create an executable from all these source files
add_executable(FOS-Antivirus WIN32
    ${MAIN_SOURCE}
    ${APP_SOURCES}
    ${ENGINE_SOURCES}
    ${UTIL_SOURCES}
    ${UI_SOURCES}
)

# 5. Tell the compiler where to look for header (.h) files
target_include_directories(FOS-Antivirus PRIVATE
    ${CMAKE_SOURCE_DIR}/src
    ${CMAKE_SOURCE_DIR}/src/app
    ${CMAKE_SOURCE_DIR}/src/engine
    ${CMAKE_SOURCE_DIR}/src/ui
    ${CMAKE_SOURCE_DIR}/src/utils
)

# --- NEW: Link GTK and SQLite to your executable ---
# This automatically handles the include paths and linker flags for GTK and SQLite
target_link_libraries(FOS-Antivirus PRIVATE
    PkgConfig::GTK
    SQLite3::SQLite3
    shlwapi
    urlmon
    wintrust
    shell32
    ole32
    oleaut32
    uuid
    advapi32
    wininet
    psapi
    bcrypt
    crypt32
)
# ----------------------------------------

# Stage the Python aggregator script next to the EXE so update_signature_db()
# can find it at runtime. Looks for it in <exe_dir>/scripts/hash_aggregator.py.
add_custom_command(TARGET FOS-Antivirus POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory
            "$<TARGET_FILE_DIR:FOS-Antivirus>/scripts"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_SOURCE_DIR}/scripts/hash_aggregator.py"
            "$<TARGET_FILE_DIR:FOS-Antivirus>/scripts/hash_aggregator.py"
    COMMENT "Staging hash aggregator script"
)

add_custom_command(TARGET FOS-Antivirus POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory
            "$<TARGET_FILE_DIR:FOS-Antivirus>/ml/models"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_SOURCE_DIR}/assets/models/forest.bin"
            "$<TARGET_FILE_DIR:FOS-Antivirus>/ml/models/forest.bin"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_SOURCE_DIR}/assets/models/forest.bin.sig"
            "$<TARGET_FILE_DIR:FOS-Antivirus>/ml/models/forest.bin.sig"
    COMMENT "Staging ML model assets (forest.bin + Ed25519 signature)"
)

# Optional: Add standard compiler flags for warnings
if(MSVC)
    target_compile_options(FOS-Antivirus PRIVATE /W3 /WX-)
else()
    target_compile_options(FOS-Antivirus PRIVATE -Wall -Wextra)
endif()

    add_executable(decision_policy_test

        src/engine/scan_executor.c
    )
    target_include_directories(decision_policy_test PRIVATE
        ${CMAKE_SOURCE_DIR}/src/engine
    )
    add_test(NAME decision_policy_test COMMAND decision_policy_test)
endif()
